echo "=== 配置 HBase 使用自带 ZooKeeper ==="

# 1. 停止 HBase
/usr/local/hbase/bin/stop-hbase.sh
sleep 5

# 2. 清理进程
pkill -9 -f HMaster
pkill -9 -f HRegionServer
pkill -9 -f HQuorumPeer

# 3. 修复 HBase 配置，使用自带 ZooKeeper
cat > /usr/local/hbase/conf/hbase-site.xml << 'EOF'
<?xml version="1.0"?>
<configuration>
  <!-- HDFS 存储路径 -->
  <property>
    <name>hbase.rootdir</name>
    <value>hdfs://node-master:9000/hbase</value>
  </property>
  
  <!-- 分布式模式 -->
  <property>
    <name>hbase.cluster.distributed</name>
    <value>true</value>
  </property>
  
  <!-- HBase 自带 ZooKeeper 配置 -->
  <property>
    <name>hbase.zookeeper.quorum</name>
    <value>node-master,node-slave1,node-slave2</value>
  </property>
  
  <property>
    <name>hbase.zookeeper.property.dataDir</name>
    <value>/usr/local/hbase/zookeeper</value>
  </property>
  
  <property>
    <name>hbase.zookeeper.property.clientPort</name>
    <value>2181</value>
  </property>
  
  <!-- Web UI 端口 -->
  <property>
    <name>hbase.master.info.port</name>
    <value>16010</value>
  </property>
  
  <!-- 其他优化配置 -->
  <property>
    <name>hbase.regionserver.handler.count</name>
    <value>30</value>
  </property>
  
  <property>
    <name>hbase.unsafe.stream.capability.enforce</name>
    <value>false</value>
  </property>
</configuration>
EOF

# 4. 分发配置
scp /usr/local/hbase/conf/hbase-site.xml asked@node-slave1:/usr/local/hbase/conf/
scp /usr/local/hbase/conf/hbase-site.xml asked@node-slave2:/usr/local/hbase/conf/

# 5. 清理 HDFS 数据
hdfs dfs -rm -r /hbase 2>/dev/null

# 6. 启动 HBase
/usr/local/hbase/bin/start-hbase.sh

sleep 15

echo "✅ HBase 配置完成，使用自带 ZooKeeper"
