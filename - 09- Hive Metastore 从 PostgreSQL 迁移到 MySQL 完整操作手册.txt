# 第五阶段：创建 Hive 元数据库和用户（master 节点）
# ============================================

# 1. 用 root 登录
mysql -u root -p
# 输密码 asdfghjkl;'

# 2. 逐条执行 SQL
CREATE DATABASE hive_metastore
  CHARACTER SET utf8
  COLLATE utf8_general_ci;

CREATE USER 'hive'@'%' IDENTIFIED WITH mysql_native_password BY "asdfghjkl;'";

GRANT ALL PRIVILEGES ON hive_metastore.* TO 'hive'@'%';

CREATE USER 'hive'@'localhost' IDENTIFIED WITH mysql_native_password BY "asdfghjkl;'";

GRANT ALL PRIVILEGES ON hive_metastore.* TO 'hive'@'localhost';

FLUSH PRIVILEGES;

# 理想输出：
# Query OK, 1 row affected, 2 warnings (0.01 sec)
# Query OK, 0 rows affected (0.02 sec)
# Query OK, 0 rows affected (0.01 sec)
# Query OK, 0 rows affected (0.00 sec)
# Query OK, 0 rows affected (0.00 sec)
# Query OK, 0 rows affected (0.01 sec)

# ⚠️ 关于 2 warnings：utf8 是 utf8mb3 的别名，MySQL 8 会提示一下，功能正常

# ⚠️ 为什么建两个用户：
# hive@'%'         → JDBC 连接匹配（Hive 通过 JDBC 连）
# hive@'localhost' → 命令行 mysql 客户端匹配（会被 MySQL 识别为 localhost）
# 少了任何一个，都会出现 Access denied

# 3. 验证

# 第七阶段：修改 hive-site.xml（master 节点）
# ============================================

# 1. 备份

# 2. 查看当前 JDO 配置
grep -n -A3 'javax.jdo' /usr/local/hive/conf/hive-site.xml

# 理想输出（改前）：
# 6:  <name>javax.jdo.option.ConnectionURL</name>
# 7-  <value>jdbc:postgresql://localhost:5432/hive</value>
# 13- <value>org.postgresql.Driver</value>
# 19- <value>hive</value>
# 25- <value>asdfghjkl;'</value>

# 3. 修改 URL 和驱动类
cd /usr/local/hive/conf

sed -i 's|jdbc:postgresql://localhost:5432/hive|jdbc:mysql://localhost:3306/hive_metastore?createDatabaseIfNotExist=true\&useSSL=false\&allowPublicKeyRetrieval=true\&characterEncoding=UTF-8\&serverTimezone=Asia/Shanghai|' hive-site.xml

sed -i 's|org.postgresql.Driver|com.mysql.cj.jdbc.Driver|' hive-site.xml

sed -i 's|PostgreSQL metastore connection URL|MySQL metastore connection URL|' hive-site.xml
sed -i 's|PostgreSQL JDBC driver class|MySQL JDBC driver class|' hive-site.xml

# 4. ⚠️ 关键修复：XML 中 & 转义
sed -i 's|&useSSL|\&amp;useSSL|g; s|&allowPublicKeyRetrieval|\&amp;allowPublicKeyRetrieval|g; s|&characterEncoding|\&amp;characterEncoding|g; s|&serverTimezone|\&amp;serverTimezone|g' hive-site.xml

# ⚠️ 必须做这一步，否则 XML 里会出现裸 &，Hive 启动时报：
# org.xml.sax.SAXParseException: The entity name must immediately follow the '&'

# 5. 验证修改

# 8-  <description>MySQL metastore connection URL</description>
# 9-  </property>
# --
# 12: <name>javax.jdo.option.ConnectionDriverName</name>

# 3. 修复 A：mapred-site.xml 加 3 个 property
nano /usr/local/hadoop/etc/hadoop/mapred-site.xml

# 在 <configuration> 内，</configuration> 之前添加：
    <property>
        <name>yarn.app.mapreduce.am.env</name>
        <value>HADOOP_MAPRED_HOME=/usr/local/hadoop</value>
    </property>
    <property>
        <name>mapreduce.map.env</name>
        <value>HADOOP_MAPRED_HOME=/usr/local/hadoop</value>
    </property>
    <property>
        <name>mapreduce.reduce.env</name>
        <value>HADOOP_MAPRED_HOME=/usr/local/hadoop</value>
    </property>

# 第十四阶段：常见问题排查
# ============================================

# 1. Access denied for user 'hive'@'localhost'
#    原因：只建了 hive@'%'，但命令行客户端被识别为 localhost
#    解决：
CREATE USER 'hive'@'localhost' IDENTIFIED WITH mysql_native_password BY "asdfghjkl;'";
GRANT ALL PRIVILEGES ON hive_metastore.* TO 'hive'@'localhost';
FLUSH PRIVILEGES;

# 2. Public Key Retrieval is not allowed
#    原因：MySQL 8 默认 caching_sha2
#    解决：用户用 mysql_native_password；或 URL 加 allowPublicKeyRetrieval=true

# 3. Unknown system variable 'query_cache_size'
#    原因：用了旧版 5.x 驱动
#    解决：换 mysql-connector-j-8.0.33.jar

# 4. The entity name must immediately follow the '&'
#    原因：XML 里裸 &
#    解决：改成 &amp;

# 5. Table already exists (schematool)
#    原因：MySQL 里已有旧 schema
#    解决：
DROP DATABASE hive_metastore;
CREATE DATABASE hive_metastore CHARACTER SET utf8 COLLATE utf8_general_ci;
#    然后重新 schematool -initSchema -dbType mysql

# 6. -> 提示符卡住
#    原因：在 mysql> 里输了 shell 命令
#    解决：输入 \c 取消，再 exit 退出

# 7. Driver class not found
#    原因：驱动没放对位置
#    解决：所有节点 /usr/local/hive/lib/ 都要有 jar

# 8. DataNode 全部 Dead
#    原因：slave 上 Hadoop 未启动
#    解决：start-dfs.sh、start-yarn.sh

# 9. MR 报 ClassNotFoundException: MRAppMaster
#    原因：mapred-site.xml 缺 HADOOP_MAPRED_HOME
#    解决：加 3 个 env property（见第十二阶段）

# 10. MR 还找不到类（加了 3 个 property 后）
#     原因：yarn-site.xml 缺 env-whitelist，容器里没传递 HADOOP_MAPRED_HOME
#     解决：加 yarn.nodemanager.env-whitelist（见第十二阶段）

# 密码：asdfghjkl;'
# 特殊字符：分号 ; 和单引号 '

# 【场景 1】交互式输入（最安全）
# 直接输，系统不解析
mysql -u root -p
# Enter password: asdfghjkl;'

# 【场景 2】SQL 里设置密码
# 用双引号包裹：末尾只留一个分号作为 SQL 结束符
ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY "asdfghjkl;'";
#                                                                            ^^ 双引号里是密码
#                                                                              ^ 双引号外的分号是 SQL 结束符

# 【场景 3】shell 变量/脚本
# 用单引号包裹，内部单引号用 '\'' 转义：
'asdfghjkl;'\'''

# 【场景 4】JDBC URL 参数
# XML 里 & 要写 &amp;
# '、; 不用转义

# 【场景 5】hive-site.xml 的 <value>
# 直接用，XML 里 '、; 是普通字符

# 【最佳实践】
# 1. 所有交互式操作都用 -p 后回车再输密码
# 2. 永远不在命令行里直接写密码
# 3. SQL 里设置密码时，末尾只留一个分号
# 4. XML 里 & 必须写成 &amp;

# 【第十七阶段】关键要点总结（共 18 条）
# ============================================
# 1. PostgreSQL 只是停用，不删除，可回退
# 2. MySQL root 密码用 mysql_native_password 插件
# 3. hive 用户要建两个：@'%' 和 @'localhost'
# 4. JDBC 驱动用 mysql-connector-j-8.0.33.jar（不是旧版 5.x）
# 5. XML 里 & 必须写 &amp;
# 6. 改完一定要 XML 校验
# 7. 3 个节点都要有驱动和配置# 8. schematool 初始化只跑一次
# 9. mapred-site.xml 必须有 3 个 HADOOP_MAPRED_HOME property
# 10. yarn-site.xml 必须有 env-whitelist
# 11. HDFS DataNode 必须先启动
# 12. HiveServer2 首次启动要 30~60 秒
# 13. Hive Metastore 切换后 HDFS warehouse 要重建
# 14. MySQL 8 建库必须指定 utf8 字符集
# 15. JDBC URL 必须加 allowPublicKeyRetrieval=true 和 serverTimezone
# 16. Ubuntu 24.04 的 MySQL 8 默认 root 用 auth_socket，需手动改密码
# 17. HiveServer2 端口是 10000，Web UI 是 10002，Metastore 是 9083
# 18. 换元数据库后，Hive 里原有的库/表信息不会自动迁移
