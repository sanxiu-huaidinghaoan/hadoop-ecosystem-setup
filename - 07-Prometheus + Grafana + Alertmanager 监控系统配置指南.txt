=============================================
Prometheus + Grafana + Alertmanager 完整监控指南
三节点大数据集群监控方案
=============================================

📋 部署前提条件
在开始部署监控系统之前，请确保以下服务已正常运行：
✓ Hadoop集群 (03号文档完成)
✓ HBase + Kafka集群 (05号文档完成) 
✓ Spark集群 (06号文档完成)
✓ 所有服务端口可访问
✓ 防火墙规则已配置
✓ 节点间网络通畅

📋 监控架构概览
Prometheus (指标收集) + Grafana (可视化) + Alertmanager (告警)
        ↑
    Node Exporter (系统指标)
    Hadoop Exporter (HDFS/YARN指标)  
    HBase Exporter (HBase指标)
    Kafka Exporter (Kafka指标)

第一阶段：基础监控组件安装（所有节点）
============================================

1. 安装 Node Exporter 1.8.2（所有节点）

# 在所有节点执行：node-master, node-slave1, node-slave2

# 下载 Node Exporter 1.8.2
cd /tmp
wget https://github.com/prometheus/node_exporter/releases/download/v1.8.2/node_exporter-1.8.2.linux-amd64.tar.gz

# 解压安装
sudo tar -xzf node_exporter-1.8.2.linux-amd64.tar.gz -C /usr/local/
sudo mv /usr/local/node_exporter-1.8.2.linux-amd64 /usr/local/node_exporter
sudo chown -R asked:asked /usr/local/node_exporter

# 创建系统服务
sudo nano /etc/systemd/system/node_exporter.service

# node_exporter.service 内容：
[Unit]
Description=Node Exporter 1.8.2
Documentation=https://prometheus.io/docs/guides/node-exporter/
After=network.target

[Service]
User=asked
Group=asked
Type=simple
ExecStart=/usr/local/node_exporter/node_exporter \
  --web.listen-address=:9100 \
  --path.rootfs=/ \
  --collector.filesystem.mount-points-exclude=^/(sys|proc|dev|host|etc)($$|/) \
  --collector.textfile.directory=/var/lib/node_exporter/textfile_collector

Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target

# 启动服务：
sudo systemctl daemon-reload
sudo systemctl enable node_exporter
sudo systemctl start node_exporter
sudo systemctl status node_exporter

# 验证
curl http://localhost:9100/metrics | head -20

2. 配置防火墙（所有节点）

sudo ufw allow 9100/tcp comment "Node Exporter"
sudo ufw status

3. 创建告警规则

mkdir -p /usr/local/prometheus/rules
nano /usr/local/prometheus/rules/cluster_alerts.yml

# cluster_alerts.yml 内容：
groups:
  - name: node_alerts
    rules:
      # 节点宕机告警
      - alert: NodeDown
        expr: up{job="node-exporter"} == 0
        for: 1m
        labels:
          severity: critical
          cluster: hadoop-cluster
        annotations:
          summary: "节点 {{ $labels.instance }} 宕机"
          description: "节点 {{ $labels.instance }} 已经超过1分钟无法访问"

      # 高CPU使用率告警
      - alert: HighCpuUsage
        expr: 100 - (avg by(instance) (rate(node_cpu_seconds_total{mode="idle"}[5m])) * 100) > 80
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "节点 {{ $labels.instance }} CPU使用率过高"
          description: "CPU使用率持续5分钟超过80%，当前值: {{ $value }}%"

      # 高内存使用率告警
      - alert: HighMemoryUsage
        expr: (1 - (node_memory_MemAvailable_bytes / node_memory_MemTotal_bytes)) * 100 > 85
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "节点 {{ $labels.instance }} 内存使用率过高"
          description: "内存使用率持续5分钟超过85%，当前值: {{ $value }}%"

      # 磁盘空间告警
      - alert: DiskSpaceLow
        expr: (1 - (node_filesystem_avail_bytes{fstype!="tmpfs"} / node_filesystem_size_bytes)) * 100 > 80
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "节点 {{ $labels.instance }} 磁盘空间不足"
          description: "分区 {{ $labels.mountpoint }} 使用率超过80%，当前值: {{ $value }}%"

  - name: hadoop_alerts
    rules:
      # HDFS 空间告警
      - alert: HDFSSpaceLow
        expr: hadoop_namenode_capacity_used_percent > 85
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "HDFS 存储空间不足"
          description: "HDFS 使用率超过85%，当前值: {{ $value }}%"

      # DataNode 宕机告警
      - alert: DataNodeDown
        expr: hadoop_namenode_num_live_data_nodes < 2
        for: 2m
        labels:
          severity: critical
        annotations:
          summary: "DataNode 节点异常"
          description: "存活的DataNode数量少于2个，当前值: {{ $value }}"

  - name: kafka_alerts
    rules:
      # Kafka Broker 宕机
      - alert: KafkaBrokerDown
        expr: kafka_server_kafkaserver_value{name="BrokerState"} != 3
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Kafka Broker {{ $labels.instance }} 状态异常"
          description: "Broker状态不是运行中(3)，当前状态: {{ $value }}"

  - name: monitoring_alerts
    rules:
      # Prometheus 自身宕机告警 - 新增
      - alert: PrometheusDown
        expr: up{job="prometheus-self"} == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Prometheus 监控服务宕机"
          description: "Prometheus 服务无法访问，监控系统可能失效"

      # Grafana 服务宕机告警 - 新增
      - alert: GrafanaDown
        expr: up{job="grafana"} == 0
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "Grafana 可视化服务异常"
          description: "Grafana 服务无法访问，仪表板可能不可用"

      # 监控数据存储空间告警 - 新增
      - alert: PrometheusStorageFull
        expr: prometheus_tsdb_storage_blocks_bytes / prometheus_tsdb_storage_blocks_bytes > 0.9
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Prometheus 存储空间不足"
          description: "Prometheus 数据存储使用率超过90%，当前值: {{ $value | humanizePercentage }}"

# 4. 创建 Prometheus 系统服务

sudo nano /etc/systemd/system/prometheus.service
# prometheus.service 内容：

[Unit]
Description=Prometheus 3.5.0 Time Series Collection and Processing Server
Documentation=https://prometheus.io/docs/
After=network.target

[Service]
User=asked
Group=asked
Type=simple
ExecStart=/usr/local/prometheus/prometheus \
  --config.file=/usr/local/prometheus/prometheus.yml \
  --storage.tsdb.path=/usr/local/prometheus/data \
  --storage.tsdb.retention.time=30d \
  --web.console.templates=/usr/local/prometheus/consoles \
  --web.console.libraries=/usr/local/prometheus/console_libraries \
  --web.listen-address=:9090 \
  --web.external-url= \
  --enable-feature=remote-write-receiver \
  --query.lookback-delta=2m

ExecReload=/bin/kill -HUP $MAINPID
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target
