组件			推荐版本		兼容范围		关键说明
Hadoop		3.3.6			3.1.0 - 3.4.x	必须使用 Hadoop 3.x
JDK			11			8, 11			推荐 JDK 11，不支持 JDK 17+
PostgreSQL	13.12		11.x - 15.x		避免使用 PostgreSQL 16+
Spark		3.3.3			3.0.x - 3.5.x	关键：匹配 Hive-Spark 版本
HBase		2.4.18		2.2.x - 2.5.x	需要对应 HBase Storage Handler
Hue			4.11.0		4.10.x - 4.12.x	需要适配 Hive 4.x 语法
Tez			0.10.2		0.9.x - 0.10.x	可选执行引擎
Zookeeper	3.8.3			3.4.x - 3.8.x	元数据高可用需要
MySQL		8.0.x			5.7.x - 8.0.x	替代 PostgreSQL 选项

PostgreSQL JDBC: 42.6.0 或 42.7.0
Apache Ranger: 2.4.0 (安全管理)
Atlas: 2.3.0 (数据治理)

# 四、系统网络配置
# ============================================

# 环境说明：
# - 操作系统：Ubuntu Server 24.04.2
# - 网络模式：NAT模式，与实体机共用路由器网络
# - 查看IP命令：ip a
# - 常见网卡名称：ens33、ens160、enp0s3、eth0

# 4.1 查看网络接口信息
ip a
# 确认网卡名称和当前IP地址

# 4.2 虚拟机网络设置参考
# - 打开 VMware → 编辑 → 虚拟网络编辑器
# - 查看 VMnet8 (NAT模式) 配置
# - 网关通常为：192.168.x.2
# - 可用IP范围：192.168.x.2-254

# 4.3 查找 Netplan 配置文件
sudo ls /etc/netplan/
# 常见配置文件：
# 00-installer-config.yaml
# 01-netcfg.yaml  
# 50-cloud-init.yaml

# 4.4 备份网络配置（重要！）
sudo cp /etc/netplan/50-cloud-init.yaml /etc/netplan/50-cloud-init.yaml.backup

# 4.5 编辑网络配置
sudo nano /etc/netplan/50-cloud-init.yaml

# Netplan 配置重要规则：
# 1. 禁止使用 Tab 键，只用空格
# 2. 每级缩进都是2个空格
# 3. 冒号 : 后面必须有一个空格
# 4. 短横线 - 后面必须有一个空格
# 5. 逗号 , 后面通常有一个空格

# 4.6 标准网络配置模板
network:                # 顶级，无缩进 - 网络配置模块
  version: 2                # 缩进2空格 - 指定netplan配置版本
  ethernets:                # 缩进2空格 - 声明下面要配置的是有线以太网接口
    ens33:                  # 缩进4空格 (2+2) - 指定要配置的网络接口名称
      dhcp4: no             # 缩进6空格 (2+2+2) - 禁用IPv4的DHCP自动获取
      dhcp6: no             # 缩进6空格 - 禁用IPv6的DHCP自动获取
      addresses:            # 缩进6空格 - 开始定义静态IP配置
        - 192.168.1.101/24 # 缩进8空格 - 设置静态IP和子网掩码
      routes:               # 缩进6空格 - 定义路由配置
        - to: 0.0.0.0/0    # 缩进8空格 - 设置默认路由
          via: 192.168.100.2 # 缩进10空格 - 指定网关地址
      nameservers:          # 缩进6空格 - 定义DNS服务器配置
        addresses: [8.8.8.8, 1.1.1.1, 192.168.1.2]  # 缩进8空格 - DNS服务器
        search: [local, home, lan]  # 缩进8空格 - DNS搜索域

# 4.7 简洁配置格式（等效）
network:
  version: 2
  ethernets:
    ens33:
      dhcp4: no
      dhcp6: no
      addresses: [192.168.100.101/24]
      routes:
        - to: default
          via: 192.168.100.2
      nameservers:
        addresses: [8.8.8.8, 1.1.1.1, 192.168.100.2]
        search: [local, home, lan]

# 4.8 保存配置
# Nano编辑器：Ctrl+X → Y → Enter

# 4.9 应用网络配置
sudo netplan apply

# 4.10 验证网络配置
ip a show ens33

# 4.11 SSH 连接配置和问题排查
# 使用本机 PowerShell 连接虚拟机
ssh asked@192.168.100.101

# SSH 连接常见问题及解决方案：

# 问题1：主机密钥变更警告（切换用户或重装系统后）
# 错误信息：WARNING: REMOTE HOST IDENTIFICATION HAS CHANGED!
# 解决方案（在 PowerShell 中执行）：
ssh-keygen -R 192.168.100.101
# 或者删除整个 known_hosts 文件（不推荐）
# Remove-Item C:\Users\YourName\.ssh\known_hosts

# 问题2：首次连接确认指纹
# 正常提示：The authenticity of host '192.168.100.101' can't be established.
# 输入 yes 接受密钥即可

# 问题3：Permission denied (publickey)
# 检查虚拟机SSH服务状态：
sudo systemctl status ssh
# 确保使用正确密码

# 4.12 优化SSH连接体验
# 在Windows主机配置SSH配置文件：
# 创建或编辑：C:\Users\YourName\.ssh\config
# 添加以下内容：
Host node-master
    HostName 192.168.100.101
    User asked
    Port 22

Host node-slave1
    HostName 192.168.100.102
    User asked
    Port 22

Host node-slave2
    HostName 192.168.100.103
    User asked
    Port 22

# 配置后可直接使用主机名连接：
ssh node-master

# ============================================

# 必须先允许SSH连接！

# 如果修改了SSH端口，使用：
# sudo ufw allow 2222/tcp

# 启用防火墙

# 查看防火墙状态

# 关闭防火墙
