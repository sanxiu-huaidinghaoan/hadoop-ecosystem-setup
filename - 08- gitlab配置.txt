节点角色分配：
- node-master:   GitLab主应用 + PostgreSQL主库 + Redis主节点 + 监控节点
- node-slave1:   GitLab备应用 + PostgreSQL从库 + Redis从节点 + 监控节点

- GitLab Web: 18000 | GitLab SSH: 2222 | PostgreSQL: 5433 | Redis: 6380
- PgBouncer: 6432 | Keepalived VIP: 192.168.100.100
- 监控端口: Node Exporter: 9100 | GitLab服务发现端口: 18000

vrrp_instance VI_1 {
    state MASTER
    interface ens33
    virtual_router_id 52
    priority 150
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass gitlab_keepalived_123
    }
    virtual_ipaddress {
        192.168.100.100/24
    }
    track_script {
        chk_gitlab
    }
}

vrrp_script chk_gitlab {
    script "/usr/bin/systemctl is-active gitlab-runsvdir"
    interval 2
    weight 50
    fall 2
    rise 2
    timeout 5
}
EOF

## 2. 创建监控告警规则
cat > /tmp/gitlab_alerts.yml << 'EOF'
groups:
  - name: gitlab_alerts
    rules:
      # GitLab 服务宕机告警
      - alert: GitLabServiceDown
        expr: up{job="gitlab-services"} == 0
        for: 2m
        labels:
          severity: critical
          cluster: gitlab-cluster
        annotations:
          summary: "GitLab 服务 {{ $labels.instance }} 宕机"
          description: "GitLab 服务 {{ $labels.instance }} 已经超过2分钟无法访问"

      # GitLab 健康状态告警
      - alert: GitLabUnhealthy
        expr: gitlab_health_status != 1
        for: 3m
        labels:
          severity: warning
        annotations:
          summary: "GitLab 服务 {{ $labels.instance }} 健康状态异常"
          description: "GitLab 健康检查失败，当前状态: {{ $value }}"

      # PostgreSQL 连接数告警
      - alert: PostgreSQLHighConnections
        expr: pg_stat_database_numbackends{datname="gitlabhq_production"} > 50
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "PostgreSQL 连接数过高"
          description: "GitLab数据库连接数超过50，当前值: {{ $value }}"

      # Redis 内存使用告警
      - alert: RedisHighMemory
        expr: redis_memory_used_bytes / redis_memory_max_bytes > 0.8
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Redis 内存使用率过高"
          description: "Redis内存使用率超过80%，当前值: {{ $value | humanizePercentage }}"

      # NFS 挂载状态告警
      - alert: NFSMountFailed
        expr: node_filesystem_device_error{mountpoint="/var/opt/gitlab/git-data"} == 1
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "NFS 存储挂载失败"
          description: "GitLab共享存储挂载异常，请立即检查"

      # VIP 故障转移告警
      - alert: VIPFailover
        expr: changes(keepalived_vrrp_state{instance=~"node-slave.*"}[5m]) > 0
        for: 0m
        labels:
          severity: info
        annotations:
          summary: "VIP 发生故障转移"
          description: "GitLab虚拟IP已从 {{ $labels.instance }} 转移"
EOF

echo "GitLab告警规则已保存到 /tmp/gitlab_alerts.yml"

cat > gitlab_health_check.sh << 'EOF'
#!/bin/bash
echo "=== GitLab 集群健康检查 ==="

echo "1. 虚拟IP状态:"
for node in 192.168.100.101 192.168.100.102 192.168.100.103; do
    echo -n "$node: "
    ssh asked@$node "ip addr show ens33 2>/dev/null | grep 192.168.100.100 || echo '未持有VIP'"
done

echo "2. GitLab服务状态:"
for node in 192.168.100.101 192.168.100.102 192.168.100.103; do
    echo "=== $node ==="
    ssh asked@$node "sudo gitlab-ctl status | grep -E 'run|down'"
done

echo "3. PostgreSQL 复制状态:"
echo "主库复制状态:"
ssh asked@192.168.100.101 "sudo -u gitlab-psql /opt/gitlab/embedded/bin/psql -h localhost -p 5433 -d gitlabhq_production -c 'SELECT client_addr, state, sync_state FROM pg_stat_replication;'"

echo "从库状态:"
ssh asked@192.168.100.102 "sudo -u gitlab-psql /opt/gitlab/embedded/bin/psql -h localhost -p 5433 -d gitlabhq_production -c 'SELECT pg_is_in_recovery();'"

echo "4. Redis 复制状态:"
echo "主Redis:"
ssh asked@192.168.100.101 "echo 'INFO replication' | sudo /opt/gitlab/embedded/bin/redis-cli -a 'asdfghjkl;''' -p 6380 | grep role"
echo "从Redis:"
ssh asked@192.168.100.102 "echo 'INFO replication' | sudo /opt/gitlab/embedded/bin/redis-cli -a 'asdfghjkl;''' -p 6380 | grep role"

echo "5. PgBouncer 连接测试:"
ssh asked@192.168.100.103 "sudo -u gitlab-psql /opt/gitlab/embedded/bin/psql -h localhost -p 6432 -d gitlabhq_production -c 'SELECT 1 AS pgbouncer_ok;'"

echo "6. NFS 挂载检查:"
for node in 192.168.100.101 192.168.100.102 192.168.100.103; do
    echo "=== $node NFS ==="
    ssh asked@$node "df -h | grep gitlab-data || echo 'NFS未挂载'"
done

echo "7. 监控集成检查:"
echo "Node Exporter状态:"
for node in 192.168.100.101 192.168.100.102 192.168.100.103; do
    status=$(curl -s -o /dev/null -w "%{http_code}" http://$node:9100/metrics)
    echo "  $node: $status"
done

echo "8. 访问测试:"
echo "虚拟IP访问:"
curl -s -o /dev/null -w "VIP响应码: %{http_code}\n" http://192.168.100.100:18000
echo "直接节点访问:"
for node in 192.168.100.101 192.168.100.102 192.168.100.103; do
    code=$(curl -s -o /dev/null -w "%{http_code}" http://$node:18000)
    echo "  $node: $code"
done

echo "9. Keepalived状态检查:"
for node in 192.168.100.101 192.168.100.102 192.168.100.103; do
    echo "=== $node ==="
